const SUITS = ['s', 'h', 'd', 'c'];
const SUIT_CHAR = { s: '♠', h: '♥', d: '♦', c: '♣' };
const RANK_STR = { 1: 'A', 11: 'J', 12: 'Q', 13: 'K' };

function makeDeck() {
  const deck = [];
  let id = 0;
  for (const suit of SUITS) {
    for (let rank = 1; rank <= 13; rank++) {
      deck.push({ id: 'c' + (id++), suit, rank, joker: false });
    }
  }
  deck.push({ id: 'c' + (id++), suit: 'j', rank: 0, joker: true });
  deck.push({ id: 'c' + (id++), suit: 'j', rank: 0, joker: true });
  return deck;
}

function shuffle(arr) {
  for (let i = arr.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [arr[i], arr[j]] = [arr[j], arr[i]];
  }
  return arr;
}

function rankStr(rank) {
  return RANK_STR[rank] || String(rank);
}

function cardValue(card, kRule) {
  if (card.joker) return 0;
  if (card.rank === 1) return 1;
  if (card.rank === 13) return kRule === 'minus1' ? -1 : 10;
  if (card.rank >= 11) return 10;
  return card.rank;
}

function handValue(hand, kRule) {
  return hand.reduce((sum, c) => sum + cardValue(c, kRule), 0);
}

/* 強カード判定（A・ジョーカー・K(−1ルール時)） */
function isStrong(card, kRule) {
  return card.joker || card.rank === 1 || (kRule === 'minus1' && card.rank === 13);
}

function isRed(card) {
  return card.suit === 'h' || card.suit === 'd';
}

/* ===== ヤニブ ルールエンジン（ホスト側で動作） ===== */

/* 捨て札の組み合わせ判定: 'single' | 'set' | 'run' | null */
function discardType(cards) {
  if (!cards || cards.length === 0) return null;
  if (cards.length === 1) return 'single';
  const jokers = cards.filter(c => c.joker);
  const normals = cards.filter(c => !c.joker);
  // 同数字セット（ジョーカーは万能札として混ぜてOK）
  if (normals.length === 0) return cards.length >= 2 ? 'set' : null;
  const sameRank = normals.every(c => c.rank === normals[0].rank);
  if (sameRank && cards.length >= 2) return 'set';
  // 階段：3枚以上・同マーク・連続（ジョーカーで穴埋め可）
  if (cards.length >= 3) {
    const sameSuit = normals.every(c => c.suit === normals[0].suit);
    if (!sameSuit) return null;
    const ranks = normals.map(c => c.rank).sort((a, b) => a - b);
    for (let i = 1; i < ranks.length; i++) if (ranks[i] === ranks[i - 1]) return null;
    let gaps = 0;
    for (let i = 1; i < ranks.length; i++) gaps += ranks[i] - ranks[i - 1] - 1;
    if (gaps <= jokers.length) return 'run';
  }
  return null;
}

/* 階段を表示順に並べる（ジョーカーを穴に配置、余りは末尾） */
function arrangeRun(cards) {
  const jokers = cards.filter(c => c.joker);
  const normals = cards.filter(c => !c.joker).sort((a, b) => a.rank - b.rank);
  const out = [normals[0]];
  let ji = 0;
  for (let i = 1; i < normals.length; i++) {
    const gap = normals[i].rank - normals[i - 1].rank - 1;
    for (let g = 0; g < gap; g++) out.push(jokers[ji++]);
    out.push(normals[i]);
  }
  while (ji < jokers.length) out.push(jokers[ji++]);
  return out;
}

class YanivEngine {
  /* players: [{id, name}], settings: {kRule, yanivMax, targetScore, halving} */
  constructor(players, settings) {
    this.players = players.map(p => ({ id: p.id, name: p.name, score: 0, out: false }));
    this.settings = settings;
    this.round = 0;
    this.phase = 'idle'; // idle | playing | roundEnd | gameEnd
    this.startIdx = 0;
  }

  newRound() {
    this.round++;
    this.deck = shuffle(makeDeck());
    this.hands = {};
    for (const p of this.players) this.hands[p.id] = this.deck.splice(0, 5);
    this.discardTop = [this.deck.shift()]; // 場の最初の1枚
    this.graveyard = [];
    this.turnIdx = this.startIdx % this.players.length;
    this.phase = 'playing';
    this.lastAction = null;
    this.roundResult = null;
  }

  currentPlayer() { return this.players[this.turnIdx]; }

  /* 直前の捨て札から取れるカード（階段なら両端のみ） */
  takeableIds() {
    const t = discardType(this.discardTop);
    if (t === 'run') {
      const ids = [this.discardTop[0].id];
      if (this.discardTop.length > 1) ids.push(this.discardTop[this.discardTop.length - 1].id);
      return ids;
    }
    return this.discardTop.map(c => c.id);
  }

  /* 手を適用。draw: {from:'deck'} or {from:'discard', cardId} */
  applyMove(pid, cardIds, draw) {
    if (this.phase !== 'playing') return { ok: false, err: 'not playing' };
    if (this.currentPlayer().id !== pid) return { ok: false, err: 'あなたの番ではありません' };
    const hand = this.hands[pid];
    const cards = cardIds.map(id => hand.find(c => c.id === id));
    if (cards.some(c => !c)) return { ok: false, err: 'カード不正' };
    const type = discardType(cards);
    if (!type) return { ok: false, err: 'その組み合わせは捨てられません' };

    let drawn;
    if (draw.from === 'deck') {
      if (this.deck.length === 0) this._reshuffle();
      drawn = this.deck.shift();
    } else {
      if (!this.takeableIds().includes(draw.cardId)) return { ok: false, err: 'そのカードは取れません' };
      const idx = this.discardTop.findIndex(c => c.id === draw.cardId);
      drawn = this.discardTop.splice(idx, 1)[0];
    }

    // 手札から除去 → 捨て札グループ更新
    this.hands[pid] = hand.filter(c => !cardIds.includes(c.id));
    this.graveyard.push(...this.discardTop);
    this.discardTop = type === 'run' ? arrangeRun(cards) : cards.slice();
    this.hands[pid].push(drawn);

    this.lastAction = {
      pid, discarded: this.discardTop.map(c => ({ ...c })),
      drewFrom: draw.from,
      tookCard: draw.from === 'discard' ? { ...drawn } : null,
    };
    this._drawnFor = { pid, cardId: drawn.id };
    this.turnIdx = (this.turnIdx + 1) % this.players.length;
    return { ok: true };
  }

  _reshuffle() {
    this.deck = shuffle(this.graveyard.splice(0));
    if (this.deck.length === 0) {
      // 極端なケース：場の札以外が尽きたら新デッキから補充
      const used = new Set([...Object.values(this.hands).flat(), ...this.discardTop].map(c => c.id));
      this.deck = shuffle(makeDeck().filter(c => !used.has(c.id)));
    }
  }

  callYaniv(pid) {
    if (this.phase !== 'playing') return { ok: false, err: 'not playing' };
    if (this.currentPlayer().id !== pid) return { ok: false, err: 'あなたの番ではありません' };
    const kRule = this.settings.kRule;
    const callerVal = handValue(this.hands[pid], kRule);
    if (callerVal > this.settings.yanivMax) return { ok: false, err: `合計${this.settings.yanivMax}以下で宣言できます` };

    // アサフ判定
    const vals = {};
    for (const p of this.players) vals[p.id] = handValue(this.hands[p.id], kRule);
    const asafers = this.players.filter(p => p.id !== pid && vals[p.id] <= callerVal);
    let winnerId, gains = {};
    if (asafers.length > 0) {
      asafers.sort((a, b) => vals[a.id] - vals[b.id]);
      winnerId = asafers[0].id;
      for (const p of this.players) {
        if (p.id === winnerId) gains[p.id] = 0;
        else if (p.id === pid) gains[p.id] = vals[p.id] + 30;
        else gains[p.id] = vals[p.id];
      }
    } else {
      winnerId = pid;
      for (const p of this.players) gains[p.id] = p.id === pid ? 0 : vals[p.id];
    }

    // スコア反映＋半減ルール
    const halved = [];
    for (const p of this.players) {
      p.score += Math.max(0, gains[p.id]);
      if (this.settings.halving === 'on' && p.score > 0 && p.score % 50 === 0) {
        p.score = p.score / 2;
        halved.push(p.id);
      }
    }

    const losers = this.players.filter(p => p.score > this.settings.targetScore);
    this.roundResult = {
      callerId: pid, winnerId, asaf: asafers.length > 0,
      hands: Object.fromEntries(this.players.map(p => [p.id, this.hands[p.id].map(c => ({ ...c }))])),
      vals, gains, halved,
      scores: Object.fromEntries(this.players.map(p => [p.id, p.score])),
      gameOver: losers.length > 0,
    };
    if (losers.length > 0) {
      this.phase = 'gameEnd';
      const sorted = [...this.players].sort((a, b) => a.score - b.score);
      this.roundResult.finalWinnerId = sorted[0].id;
      this.roundResult.finalLoserId = sorted[sorted.length - 1].id;
    } else {
      this.phase = 'roundEnd';
      this.startIdx = this.players.findIndex(p => p.id === winnerId);
    }
    return { ok: true };
  }

  /* 各プレイヤー専用の状態（他人の手札は枚数のみ！） */
  stateFor(pid) {
    const kRule = this.settings.kRule;
    const s = {
      type: 'state',
      phase: this.phase,
      round: this.round,
      settings: this.settings,
      youId: pid,
      yourHand: (this.hands && this.hands[pid]) ? this.hands[pid].map(c => ({ ...c })) : [],
      players: this.players.map((p, i) => ({
        id: p.id, name: p.name, score: p.score,
        cardCount: (this.hands && this.hands[p.id]) ? this.hands[p.id].length : 0,
        isTurn: this.phase === 'playing' && i === this.turnIdx,
      })),
      discardTop: this.discardTop ? this.discardTop.map(c => ({ ...c })) : [],
      takeableIds: this.phase === 'playing' ? this.takeableIds() : [],
      deckCount: this.deck ? this.deck.length : 0,
      turnId: this.phase === 'playing' ? this.currentPlayer().id : null,
      yanivMax: this.settings.yanivMax,
      yourValue: (this.hands && this.hands[pid]) ? handValue(this.hands[pid], kRule) : 0,
      lastAction: this.lastAction,
      roundResult: this.roundResult,
    };
    // 直前に自分が引いたカードだけ演出用に伝える
    if (this._drawnFor && this._drawnFor.pid === pid) s.justDrew = this._drawnFor.cardId;
    return s;
  }
}

/* ===== PeerJS 通信（部屋番号マッチング） ===== */
const Net = {
  peer: null,
  isHost: false,
  roomCode: null,
  conns: new Map(),   // ホスト用: peerId -> conn
  hostConn: null,     // ゲスト用
  onData: null,       // (fromId, msg) => {}
  onJoin: null,       // ホスト用 (peerId) => {}
  onLeave: null,      // (peerId) => {}

  _peerId(code) { return 'yaniv-jp-room-' + code; },

  createRoom(onReady, onError) {
    this.isHost = true;
    const code = String(Math.floor(100000 + Math.random() * 900000));
    const peer = new Peer(this._peerId(code), { debug: 1 });
    this.peer = peer;
    peer.on('open', () => {
      this.roomCode = code;
      onReady(code);
    });
    peer.on('error', (err) => {
      if (err.type === 'unavailable-id') {
        peer.destroy();
        this.createRoom(onReady, onError); // 番号衝突 → 再生成
      } else {
        onError(err);
      }
    });
    peer.on('connection', (conn) => {
      conn.on('open', () => {
        this.conns.set(conn.peer, conn);
        if (this.onJoin) this.onJoin(conn.peer);
      });
      conn.on('data', (msg) => { if (this.onData) this.onData(conn.peer, msg); });
      conn.on('close', () => {
        this.conns.delete(conn.peer);
        if (this.onLeave) this.onLeave(conn.peer);
      });
    });
  },

  joinRoom(code, onReady, onError) {
    this.isHost = false;
    const peer = new Peer({ debug: 1 });
    this.peer = peer;
    let opened = false;
    peer.on('open', () => {
      const conn = peer.connect(this._peerId(code), { reliable: true });
      const timeout = setTimeout(() => {
        if (!opened) onError(new Error('部屋が見つかりません（番号を確認してね）'));
      }, 8000);
      conn.on('open', () => {
        opened = true;
        clearTimeout(timeout);
        this.hostConn = conn;
        this.roomCode = code;
        onReady();
      });
      conn.on('data', (msg) => { if (this.onData) this.onData('host', msg); });
      conn.on('close', () => { if (this.onLeave) this.onLeave('host'); });
    });
    peer.on('error', (err) => {
      if (err.type === 'peer-unavailable') onError(new Error('部屋が見つかりません（番号を確認してね）'));
      else onError(err);
    });
  },

  /* ホスト → 特定ゲスト */
  sendTo(peerId, msg) {
    const conn = this.conns.get(peerId);
    if (conn && conn.open) conn.send(msg);
  },

  /* ゲスト → ホスト */
  sendToHost(msg) {
    if (this.hostConn && this.hostConn.open) this.hostConn.send(msg);
  },

  /* ホスト → 全ゲスト */
  broadcast(msg) {
    for (const conn of this.conns.values()) if (conn.open) conn.send(msg);
  },
};

      ctx.strokeStyle = 'rgba(255,230,150,.55)'; ctx.lineWidth = 3; ctx.stroke();
      // 数字
      ctx.save();
      ctx.translate(cx, cy);
      ctx.rotate(a0 + arc / 2);
      ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
      ctx.fillStyle = '#fff';
      ctx.font = 'bold 64px "Noto Sans JP", sans-serif';
      ctx.shadowColor = 'rgba(0,0,0,.5)'; ctx.shadowBlur = 8;
      ctx.fillText(String(segs[i]), R * 0.68, 0);
      ctx.restore();
    }
    // 中央キャップ
    ctx.beginPath(); ctx.arc(cx, cy, 44, 0, Math.PI * 2);
    const cap = ctx.createRadialGradient(cx - 10, cy - 10, 5, cx, cy, 44);
    cap.addColorStop(0, '#f5e3a0'); cap.addColorStop(1, '#9a7a22');
    ctx.fillStyle = cap; ctx.fill();
    ctx.fillStyle = '#5c4512';
    ctx.font = 'bold 30px serif'; ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
    ctx.fillText('🍺', cx, cy + 2);
  },

  /* resultIdx に止まるよう回転。onDone(drinks) */
  spin(resultIdx, onDone) {
    if (this.spinning) return;
    this.spinning = true;
    const segs = WHEELS[this.wheelIdx].segs;
    const n = segs.length;
    const arc = (Math.PI * 2) / n;
    // ポインタは真上(-90°基準で描いているので angle=0 のとき seg0 が右上から始まる)
    // seg i の中心を真上に：angle = -(i*arc + arc/2)
    const jitter = (Math.random() - 0.5) * arc * 0.6;
    const target = -(resultIdx * arc + arc / 2) + jitter;
    const turns = 5 + Math.floor(Math.random() * 3);
    const final = target - turns * Math.PI * 2;
    const start = this.angle;
    const dist = final - start;
    const dur = 4200 + Math.random() * 800;
    const t0 = performance.now();
    let lastTick = -1;
    const step = (now) => {
      const t = Math.min(1, (now - t0) / dur);
      const ease = 1 - Math.pow(1 - t, 3); // cubic out
      this.angle = start + dist * ease;
      this.draw(this.angle);
      // コトコト音
      const tick = Math.floor(Math.abs(this.angle) / arc);
      if (tick !== lastTick) { Sfx.spin(); lastTick = tick; }
      if (t < 1) requestAnimationFrame(step);
      else {
        this.spinning = false;
        onDone(segs[resultIdx]);
      }
    };
    requestAnimationFrame(step);
  },
};

</script>

    Net.broadcast({ type: 'backToLobby' });
    resetToLobby();
  };

  function resetToLobby() {
    App.engine = null;
    App.prevRound = 0;
    App.selected.clear();
    if (App.isHost) hostBroadcastLobby();
    enterLobby(Net.roomCode);
  }

  function esc(s) {
    return String(s).replace(/[&<>"']/g, c => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]));
  }
})();
